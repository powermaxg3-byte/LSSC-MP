    function isAppleDevice() {

    /*************************************************************************

     * 3. MAP URL BUILDER (Supports Apple Maps, Google Maps, & Custom URLs)

     *************************************************************************/

    function buildMapUrl(item) {

      const provider = getResolvedProvider();

      const label = encodeURIComponent(`NS ${item.mp} - ${item.station}`);

      const lat = item.lat.toFixed(6);

      const lng = item.lng.toFixed(6);


      // If a custom URL override is saved, check if user wants cross-platform coordinate conversion

      if (item.customUrl && item.customUrl.trim() !== '') {

        const rawUrl = item.customUrl.trim();

        // Extract coordinates if embedded in an Apple/Google URL so both iOS and Android crews can use it

        const coordMatch = rawUrl.match(/([-+]?\d{1,2}\.\d+)\s*,\s*([-+]?\d{1,3}\.\d+)/);

        if (coordMatch) {

          const cLat = coordMatch[1];

          const cLng = coordMatch[2];

          if (provider === 'apple') {

            return `https://maps.apple.com/?ll=${cLat},${cLng}&q=${label}&t=h`;

          } else {

            return `https://www.google.com/maps/search/?api=1&query=${cLat},${cLng}`;

          }

        }

        return rawUrl;

      }


      if (provider === 'apple') {

        // Hybrid satellite + road map view ideal for rail corridors

        return `https://maps.apple.com/?ll=${lat},${lng}&q=${label}&t=h`;

      } else {

        return `https://www.google.com/maps/search/?api=1&query=${lat},${lng}`;

      }

    }


    /*************************************************************************

     * 4. DISTANCE CALCULATION (Haversine in Miles for "Find Nearest")

     *************************************************************************/

    function getDistanceMiles(lat1, lon1, lat2, lon2) {

      const R = 3958.8; // Earth radius in miles

      const dLat = ((lat2 - lat1) * Math.PI) / 180;

      const dLon = ((lon2 - lon1) * Math.PI) / 180;

      const a =

        Math.sin(dLat / 2) * Math.sin(dLat / 2) +

        Math.cos((lat1 * Math.PI) / 180) *

          Math.cos((lat2 * Math.PI) / 180) *

          Math.sin(dLon / 2) *

          Math.sin(dLon / 2);

      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

      return R * c;

    }

    function togglePin(id) {

      if (pinnedIds.includes(id)) {

        pinnedIds = pinnedIds.filter(item => item !== id);

        showToast('Removed from Pinned mileposts');

      } else {

        pinnedIds.push(id);

    function locateNearestMilepost() {

      if (!navigator.geolocation) {

        showToast('Geolocation is not supported by your browser.');

        return;

      }


      const btnText = document.getElementById('locate-btn-text');

      btnText.textContent = 'Acquiring GPS Signal...';


      navigator.geolocation.getCurrentPosition(

        pos => {

          userLocation = {

            lat: pos.coords.latitude,

            lng: pos.coords.longitude

          };

          btnText.textContent = 'Update Nearest GPS';

          document.getElementById('gps-status-banner').classList.remove('hidden');

          activePageFilter = 'all';

          renderMileposts();

          showToast('Sorted mileposts by closest distance to you!');

        },

        err => {

          btnText.textContent = 'Find Nearest Milepost to Me';

          showToast('Unable to read GPS. Check location permissions.');

        },

        { enableHighAccuracy: true, timeout: 10000 }

      );

    }

    // Reliable Clipboard Helper for iFrames and Mobile WebViews

    function copyTextToClipboard(text, successMsg) {

      const textarea = document.createElement('textarea');

      textarea.value = text;

      textarea.style.position = 'fixed';

      textarea.style.opacity = '0';

      document.body.appendChild(textarea);

      textarea.select();

      try {

        document.execCommand('copy');

        showToast(successMsg || 'Copied to clipboard!');

      } catch (e) {

        showToast('Copy failed');

      }

      document.body.removeChild(textarea);

    }

    function saveMpOverride(e) {

      e.preventDefault();

      const lat = parseFloat(document.getElementById('edit-mp-lat').value);

      const lng = parseFloat(document.getElementById('edit-mp-lng').value);

      closeEditMpModal();
